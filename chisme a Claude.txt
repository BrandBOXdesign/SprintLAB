// ============================================================
// NUEVO: SEGUIMIENTO PESADO (HEAVY TRACKING)
// ============================================================

/** Umbrales y parámetros (ajustables) */
const TRACKING_CONFIG = {
  ALPHA: 0.62,                // suavizado EMA (0.5-0.75)
  MAX_SPEEDS: {               // metros/segundo máximos por articulación
    27: 6.8, 28: 6.8,         // tobillos
    25: 5.8, 26: 5.8,         // rodillas
    23: 3.8, 24: 3.8,         // caderas
    11: 4.0, 12: 4.0,         // hombros
    13: 5.0, 14: 5.0,         // codos
    15: 6.0, 16: 6.0          // muñecas
  },
  SPEED_MULTIPLIER: 1.65,     // factor para considerar outlier
  SWAP_DIST_THRESHOLD: 0.28,  // metros (distancia 3D entre puntos simétricos)
  FEMUR_REF_FRAMES: 10        // primeros N frames para calcular fémur de referencia
};

// ------------------------------------------------------------
// 1. TRACKER POR ARTICULACIÓN (EMA + Outlier)
// ------------------------------------------------------------
class JointTracker {
  constructor() {
    this.history = new Map();
    this.ALPHA = TRACKING_CONFIG.ALPHA;
    this.MAX_SPEEDS = TRACKING_CONFIG.MAX_SPEEDS;
    this.SPEED_MULT = TRACKING_CONFIG.SPEED_MULTIPLIER;
  }

  update(idx, world, t) {
    if (!world) return null;
    const key = idx.toString();
    if (!this.history.has(key)) {
      this.history.set(key, { world: { ...world }, t, vel: { x: 0, y: 0, z: 0 } });
      return { ...world, suspect: false };
    }

    const prev = this.history.get(key);
    const dt = Math.max(0.001, t - prev.t);

    // Predicción con velocidad constante
    const pred = {
      x: prev.world.x + prev.vel.x * dt,
      y: prev.world.y + prev.vel.y * dt,
      z: prev.world.z + prev.vel.z * dt
    };

    // Velocidad observada
    const speed = Math.hypot(world.x - pred.x, world.y - pred.y, world.z - pred.z) / dt;
    const maxSp = this.MAX_SPEEDS[idx] || 4.5;
    let accepted = world;
    let suspect = false;

    if (speed > maxSp * this.SPEED_MULT) {
      // Outlier: usamos la predicción
      accepted = { ...pred };
      suspect = true;
    }

    // EMA suavizado
    const smoothed = {
      x: prev.world.x * this.ALPHA + accepted.x * (1 - this.ALPHA),
      y: prev.world.y * this.ALPHA + accepted.y * (1 - this.ALPHA),
      z: prev.world.z * this.ALPHA + accepted.z * (1 - this.ALPHA),
      suspect
    };

    // Actualizar velocidad
    const newVel = {
      x: (smoothed.x - prev.world.x) / dt,
      y: (smoothed.y - prev.world.y) / dt,
      z: (smoothed.z - prev.world.z) / dt
    };

    this.history.set(key, { world: { ...smoothed }, t, vel: newVel });
    return { ...smoothed };
  }
}

// 2. CORRECCIÓN DE SWAPS (IZQUIERDA/DERECHA)
// ------------------------------------------------------------
function correctIdentitySwaps(frames) {
  const pairs = [
    [23, 24], // caderas
    [25, 26], // rodillas
    [27, 28], // tobillos
    [11, 12], // hombros
    [13, 14], // codos
    [15, 16]  // muñecas
  ];

  for (let i = 1; i < frames.length; i++) {
    const curr = frames[i];
    const prev = frames[i - 1];
    const currPts = curr.pts;
    const prevPts = prev.pts;

    for (const [lIdx, rIdx] of pairs) {
      const l = currPts[lIdx]?.world;
      const r = currPts[rIdx]?.world;
      if (!l || !r) continue;

      const dist = dist3D(l, r);
      if (dist < TRACKING_CONFIG.SWAP_DIST_THRESHOLD) {
        // Criterio de decisión: comparar la distancia de cada punto actual
        // con la posición previa del mismo lado (continuidad)
        const prevL = prevPts[lIdx]?.world;
        const prevR = prevPts[rIdx]?.world;
        if (!prevL || !prevR) continue;

        const dLL = dist3D(l, prevL);
        const dRL = dist3D(r, prevL);
        const dRR = dist3D(r, prevR);
        const dLR = dist3D(l, prevR);

        // Si el punto derecho está más cerca de la posición previa izquierda
        // que el punto izquierdo, y viceversa, hay swap
        if (dRL < dLL && dLR < dRR) {
          // Intercambiamos etiquetas
          swapLabels(curr, lIdx, rIdx);
          currPts[lIdx].suspect = true;
          currPts[rIdx].suspect = true;
        }
      }
    }
  }
}

/** Intercambia las etiquetas de dos puntos en un frame (tanto 2D como world) */
function swapLabels(frame, idxA, idxB) {
  const pts = frame.pts;
  // Intercambiar objetos completos
  const tmp = pts[idxA];
  pts[idxA] = pts[idxB];
  pts[idxB] = tmp;
  // Asegurar que el marcador suspect se mantenga en ambos
  if (pts[idxA]) pts[idxA].suspect = true;
  if (pts[idxB]) pts[idxB].suspect = true;
}

/** Calcula la longitud de fémur de referencia a partir de los primeros frames */
function calculateReferenceFemur(frames) {
  const femurs = [];
  const limit = Math.min(frames.length, TRACKING_CONFIG.FEMUR_REF_FRAMES);
  for (let i = 0; i < limit; i++) {
    const hipL = frames[i].pts[23]?.world;
    const kneeL = frames[i].pts[25]?.world;
    const hipR = frames[i].pts[24]?.world;
    const kneeR = frames[i].pts[26]?.world;
    if (hipL && kneeL) {
      femurs.push(dist3D(hipL, kneeL));
    }
    if (hipR && kneeR) {
      femurs.push(dist3D(hipR, kneeR));
    }
  }
  if (femurs.length === 0) return null;
  // Mediana para ser robusto
  femurs.sort((a, b) => a - b);
  return femurs[Math.floor(femurs.length / 2)];
}

/** Ángulo entre dos vectores (en grados) */
function angleBetweenVectors(u, v) {
  const dot = u.x * v.x + u.y * v.y + u.z * v.z;
  const nu = Math.hypot(u.x, u.y, u.z) || 1;
  const nv = Math.hypot(v.x, v.y, v.z) || 1;
  let cos = dot / (nu * nv);
  cos = Math.max(-1, Math.min(1, cos));
  return Math.acos(cos) * (180 / Math.PI);
}

/** Ángulo de flexión de rodilla (0° = extendida, 180° = flexionada máxima) */
function computeKneeAngle(hip, knee, ankle) {
  if (!hip || !knee || !ankle) return null;
  const u = { x: knee.x - hip.x, y: knee.y - hip.y, z: knee.z - hip.z };
  const v = { x: ankle.x - knee.x, y: ankle.y - knee.y, z: ankle.z - knee.z };
  const theta = angleBetweenVectors(u, v);
  return 180 - theta; // convención: 180° = extendida
}

/** Ángulo de cadera (0° = extensión, 180° = flexión máxima) */
function computeHipAngle(shoulder, hip, knee) {
  if (!shoulder || !hip || !knee) return null;
  const u = { x: shoulder.x - hip.x, y: shoulder.y - hip.y, z: shoulder.z - hip.z };
  const v = { x: knee.x - hip.x, y: knee.y - hip.y, z: knee.z - hip.z };
  const theta = angleBetweenVectors(u, v);
  return 180 - theta;
}

/** Inclinación de tronco respecto a la vertical (0° = vertical, positivo hacia adelante) */
function computeTrunkInclination(shoulder, hip) {
  if (!shoulder || !hip) return null;
  const u = { x: shoulder.x - hip.x, y: shoulder.y - hip.y, z: shoulder.z - hip.z };
  const vertical = { x: 0, y: 1, z: 0 };
  return angleBetweenVectors(u, vertical);
}

/** Ángulo de codo (180° = extendido) */
function computeElbowAngle(shoulder, elbow, wrist) {
  if (!shoulder || !elbow || !wrist) return null;
  const u = { x: elbow.x - shoulder.x, y: elbow.y - shoulder.y, z: elbow.z - shoulder.z };
  const v = { x: wrist.x - elbow.x, y: wrist.y - elbow.y, z: wrist.z - elbow.z };
  const theta = angleBetweenVectors(u, v);
  return 180 - theta;
}

/** Distancia 3D */
function dist3D(a, b) {
  return Math.hypot(a.x - b.x, a.y - b.y, a.z - b.z);
}

/** Altura de rodilla normalizada (%) respecto a la longitud del fémur */
function computeKneeHeightNormalized(hip, knee, femurLength) {
  if (!hip || !knee || !femurLength || femurLength === 0) return null;
  const height = hip.y - knee.y; // positivo si rodilla más baja
  return (height / femurLength) * 100;
}

/** Calcula todas las métricas para un frame */
function computeFrameMetrics(frame, femurLength) {
  const pts = frame.pts;
  const getW = (idx) => pts[idx]?.world || null;

  const hipL = getW(23), kneeL = getW(25), ankleL = getW(27);
  const hipR = getW(24), kneeR = getW(26), ankleR = getW(28);
  const shoulderL = getW(11), shoulderR = getW(12);
  const elbowL = getW(13), elbowR = getW(14);
  const wristL = getW(15), wristR = getW(16);

  return {
    kneeL: computeKneeAngle(hipL, kneeL, ankleL),
    kneeR: computeKneeAngle(hipR, kneeR, ankleR),
    hipL: computeHipAngle(shoulderL, hipL, kneeL),
    hipR: computeHipAngle(shoulderR, hipR, kneeR),
    trunk: computeTrunkInclination(
      // Promedio de hombros y caderas para el tronco
      shoulderL && shoulderR ? { x: (shoulderL.x + shoulderR.x)/2, y: (shoulderL.y + shoulderR.y)/2, z: (shoulderL.z + shoulderR.z)/2 } : null,
      hipL && hipR ? { x: (hipL.x + hipR.x)/2, y: (hipL.y + hipR.y)/2, z: (hipL.z + hipR.z)/2 } : null
    ),
    elbowL: computeElbowAngle(shoulderL, elbowL, wristL),
    elbowR: computeElbowAngle(shoulderR, elbowR, wristR),
    kneeHeightNormL: computeKneeHeightNormalized(hipL, kneeL, femurLength),
    kneeHeightNormR: computeKneeHeightNormalized(hipR, kneeR, femurLength),
  };
}

// ------------------------------------------------------------
// 4. FUNCIÓN PRINCIPAL: applyHeavyTracking
// ------------------------------------------------------------
function applyHeavyTracking(frames) {
  if (frames.length < 3) return;

  // 1. Suavizado + outlier por articulación
  const tracker = new JointTracker();
  for (let i = 0; i < frames.length; i++) {
    const frame = frames[i];
    const pts = frame.pts;
    for (const idx of SKELETON_LANDMARKS) {
      const p = pts[idx];
      if (p?.world) {
        const updated = tracker.update(parseInt(idx), p.world, frame.t);
        if (updated) {
          p.world = updated;         // reemplazamos con el suavizado
          if (updated.suspect) p.suspect = true;
        }
      }
    }
  }

  // 2. Corrección de swaps (utiliza el frame anterior)
  correctIdentitySwaps(frames);

  // 3. Cálculo de métricas
  const femurRef = calculateReferenceFemur(frames);
  for (const frame of frames) {
    frame.metrics = computeFrameMetrics(frame, femurRef);
  }
}

function correctIdentitySwaps(frames) {
  const pairs = [[23,24],[25,26],[27,28],[11,12],[13,14],[15,16]];
  const decisions = new Array(frames.length).fill(null).map(() => ({}));

  // ---- PASADA HACIA ADELANTE ----
  for (let i = 1; i < frames.length; i++) {
    const curr = frames[i].pts;
    const prev = frames[i-1].pts;
    for (const [lIdx, rIdx] of pairs) {
      const l = curr[lIdx]?.world;
      const r = curr[rIdx]?.world;
      if (!l || !r) continue;
      if (dist3D(l, r) > TRACKING_CONFIG.SWAP_DIST_THRESHOLD) continue;

      const prevL = prev[lIdx]?.world;
      const prevR = prev[rIdx]?.world;
      if (!prevL || !prevR) continue;

      const dLL = dist3D(l, prevL);
      const dRL = dist3D(r, prevL);
      const dRR = dist3D(r, prevR);
      const dLR = dist3D(l, prevR);

      // Si la asignación cruzada es más coherente, marcamos swap
      if (dRL < dLL && dLR < dRR) {
        decisions[i][lIdx] = decisions[i][rIdx] = true;
      }
    }
  }

  // ---- PASADA HACIA ATRÁS ----
  for (let i = frames.length - 2; i >= 0; i--) {
    const curr = frames[i].pts;
    const next = frames[i+1].pts;
    for (const [lIdx, rIdx] of pairs) {
      const l = curr[lIdx]?.world;
      const r = curr[rIdx]?.world;
      if (!l || !r) continue;
      if (dist3D(l, r) > TRACKING_CONFIG.SWAP_DIST_THRESHOLD) continue;

      const nextL = next[lIdx]?.world;
      const nextR = next[rIdx]?.world;
      if (!nextL || !nextR) continue;

      const dLL = dist3D(l, nextL);
      const dRL = dist3D(r, nextL);
      const dRR = dist3D(r, nextR);
      const dLR = dist3D(l, nextR);

      if (dRL < dLL && dLR < dRR) {
        // Si ya estaba marcado en forward, confirmamos; si no, lo marcamos también
        if (decisions[i][lIdx] !== undefined) {
          decisions[i][lIdx] = decisions[i][rIdx] = true;
        } else {
          // Solo aplicamos si también lo confirma el forward (para evitar falsos positivos)
          // Pero para ser conservadores, lo marcamos solo si el forward también lo ve.
        }
      }
    }
  }

  // ---- APLICAR SOLO DONDE HAY CONSENSO (forward && backward) ----
  for (let i = 1; i < frames.length - 1; i++) {
    const curr = frames[i];
    for (const [lIdx, rIdx] of pairs) {
      // Verificar si tanto forward (i) como backward (i) lo marcaron
      const fwd = decisions[i] && (decisions[i][lIdx] === true);
      // Backward: la decisión en i también se almacena en decisions[i]
      // Como hicimos dos pasadas, decisions[i] tiene la info combinada.
      // Para simplificar, aplicamos si decisions[i][lIdx] es true (ya que backward reafirma)
      if (fwd) {
        // Doble verificación: ¿sigue siendo válido tras los swaps previos?
        const l = curr.pts[lIdx]?.world;
        const r = curr.pts[rIdx]?.world;
        if (!l || !r) continue;
        const prev = frames[i-1].pts;
        const next = frames[i+1].pts;
        const prevL = prev[lIdx]?.world;
        const prevR = prev[rIdx]?.world;
        const nextL = next[lIdx]?.world;
        const nextR = next[rIdx]?.world;
        if (!prevL || !prevR || !nextL || !nextR) continue;

        // Si las distancias cruzadas son consistentes en ambos lados, aplicamos
        const dRL_prev = dist3D(r, prevL);
        const dLL_prev = dist3D(l, prevL);
        const dLR_next = dist3D(l, nextR);
        const dRR_next = dist3D(r, nextR);

        if (dRL_prev < dLL_prev && dLR_next < dRR_next) {
          swapLabels(curr, lIdx, rIdx);
          curr.pts[lIdx].suspect = true;
          curr.pts[rIdx].suspect = true;
        }
      }
    }
  }
}

class JointTracker {
  constructor() {
    this.history = new Map();
    this.ALPHA = TRACKING_CONFIG.ALPHA;
    this.MAX_JUMP = TRACKING_CONFIG.MAX_POSITION_CHANGE;
  }

  update(idx, world, t) {
    if (!world) return null;
    const key = idx.toString();
    if (!this.history.has(key)) {
      this.history.set(key, { world: { ...world }, t });
      return { ...world, suspect: false };
    }

    const prev = this.history.get(key);
    const dt = Math.max(0.001, t - prev.t);

    // SALTO ABSOLUTO: distancia entre la observación actual y la posición suavizada anterior
    const jump = dist3D(world, prev.world);
    const maxJump = this.MAX_JUMP[idx] || 0.40;

    let accepted = world;
    let suspect = false;

    if (jump > maxJump) {
      // Outlier: usamos la posición anterior (congelamos) en lugar de extrapolar
      // Esto evita penalizar aceleraciones/frenadas, solo captura glitches.
      accepted = { ...prev.world };
      suspect = true;
    }

    // EMA suavizado
    const smoothed = {
      x: prev.world.x * this.ALPHA + accepted.x * (1 - this.ALPHA),
      y: prev.world.y * this.ALPHA + accepted.y * (1 - this.ALPHA),
      z: prev.world.z * this.ALPHA + accepted.z * (1 - this.ALPHA),
      suspect: suspect
    };

    this.history.set(key, { world: { ...smoothed }, t });
    return { ...smoothed };
  }
}

function computeKneeHeightNormalized(hip, knee, femurLength) {
  if (!hip || !knee || !femurLength || femurLength === 0) return null;
  // Y-up: si knee.y > hip.y, la rodilla está más arriba (físicamente imposible)

function drawSkeleton(ctx, connections, frame, canvasW, canvasH, contentRect) {
  ctx.clearRect(0, 0, canvasW, canvasH);
  if (!frame) return;

  const { x: ox, y: oy, w, h } = contentRect;
